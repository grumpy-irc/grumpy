PROJECT(irc2htmlcode_ut)

SET(QT_USE_QTTEST TRUE)
SET(CMAKE_BUILD_TYPE Release)
SET(CMAKE_AUTOMOC ON)

option(QT6_BUILD "Build with Qt6" false)

if(QT6_BUILD)
  find_package(Qt6Core REQUIRED)
  find_package(Qt6Test REQUIRED)
  set(QT_INCLUDES ${Qt6Test_INCLUDE_DIRS})
  include_directories(${QT_INCLUDES})
elseif(QT5_BUILD)
  find_package(Qt5Core REQUIRED)
  find_package(Qt5Test REQUIRED)
  set(QT_INCLUDES ${Qt5Test_INCLUDE_DIRS})
  include_directories(${QT_INCLUDES})
else()
  message(FATAL_ERROR "Enable QT5_BUILD or QT6_BUILD")
endif()

file (GLOB headers "*.h")
file (GLOB src "*.cpp")
file (GLOB resources "*.qrc")

if(QT6_BUILD)
    QT6_WRAP_CPP(irc2htmlcode_ut_HEADERS_MOC ${irc2htmlcode_ut_HEADERS})
    QT6_ADD_RESOURCES(irc2htmlcode_ut_RESOURCES_RCC ${resources})
elseif(QT5_BUILD)
    QT5_WRAP_CPP(irc2htmlcode_ut_HEADERS_MOC ${irc2htmlcode_ut_HEADERS})
    QT5_ADD_RESOURCES(irc2htmlcode_ut_RESOURCES_RCC ${resources})
else()
    message(FATAL_ERROR "Enable QT5_BUILD or QT6_BUILD")
endif()

ADD_DEFINITIONS(${QT_DEFINITIONS})
ADD_DEFINITIONS( -DQT_USE_QSTRINGBUILDER )

ADD_EXECUTABLE(irc2htmlcode_ut ${src} ${irc2htmlcode_ut_HEADERS_MOC} ${irc2htmlcode_ut_RESOURCES_RCC} ${headers})
if (QT6_BUILD)
    TARGET_LINK_LIBRARIES(irc2htmlcode_ut Qt6::Core Qt6::Test)
elseif (QT5_BUILD)
    TARGET_LINK_LIBRARIES(irc2htmlcode_ut Qt5::Core Qt5::Test)
endif()
TARGET_LINK_LIBRARIES(irc2htmlcode_ut irc2htmlcode ${QT_LIBRARIES})
INCLUDE_DIRECTORIES(${CMAKE_CURRENT_BINARY_DIR})

add_test(NAME irc2htmlcode_ut COMMAND irc2htmlcode_ut)
